Developer first

One API call.
Message sent.

Simple, predictable REST. Full reference with examples in five languages in the docs.

send-message — NecxaWA API
# Send a WhatsApp text message
curl -X POST http://localhost:3000/api/sessions/{sessionId}/messages/send-text \
  -H "x-api-key: YOUR_API_KEY" \
  -H "Content-Type: application/json" \
  -d '{"to": "919876543210", "text": "Hello from NecxaWA!"}'
await fetch(`http://localhost:3000/api/sessions/${sessionId}/messages/send-text`, {
  method: 'POST',
  headers: { 'x-api-key': API_KEY, 'Content-Type': 'application/json' },
  body: JSON.stringify({ to: '919876543210', text: 'Hello from NecxaWA!' })
});
import requests

requests.post(
  f"http://localhost:3000/api/sessions/{session_id}/messages/send-text",
  headers={"x-api-key": API_KEY},
  json={"to": "919876543210", "text": "Hello from NecxaWA!"},
)
$ch = curl_init("http://localhost:3000/api/sessions/$sessionId/messages/send-text");
curl_setopt($ch, CURLOPT_HTTPHEADER, ["x-api-key: $apiKey", "Content-Type: application/json"]);
curl_setopt($ch, CURLOPT_POSTFIELDS, json_encode(['to' => '919876543210', 'text' => 'Hello from NecxaWA!']));
